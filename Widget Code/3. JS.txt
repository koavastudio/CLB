    {   // 2: big orange back with a black spot, orange head patch, black-and-orange striped tail
      torso: function (nx, ny, P) { var k = cBlob(nx, ny, P, 0.42, 0.2, 0.13, 0.16, 0.06, 0); return k > 0.01 ? [2, k] : [1, cBlob(nx, ny, P, 0.4, 0.04, 0.36, 0.5, 0.05, 1.5)]; },
      head: function (nx, ny, P) { return [1, cBlob(nx, ny, P, 0.26, 0.3, 0.32, 0.34, 0.07, 3)]; },
      tail: function (nx, ny, P) { return ny < 0.36 ? [2, cEdge((0.36 - ny) * P.h)] : [1, cEdge((ny - 0.36) * P.h)]; }
    }
  ];
  // which coat colour (0 = base, else index into the pattern colours) and how much, at a rig-space point of part n
  function patternAt(coat, n, px_, py_, wx, wy, blobs) {
    var P = C.parts[n] || C.parts.torso, nx = px_ / P.w, ny = py_ / P.h;
    var head = /^head|^face/.test(n), leg = /^leg/.test(n);
    if (coat.stripe) {
      var v, k;
      if (head) { v = Math.sin(nx * 30 + 0.6); k = smooth(0.35, 0.85, v) * smooth(0.48, 0.2, ny) * smooth(0.12, 0.3, nx) * smooth(0.82, 0.6, nx); }
      else if (leg) { v = Math.sin(py_ * 0.12 + 0.5); k = smooth(0.4, 0.85, v) * smooth(0.82, 0.6, ny) * smooth(0.05, 0.25, ny); }
      else if (n === 'tail') { v = Math.sin((px_ * 0.5 + py_) * 0.085); k = smooth(0.35, 0.8, v); }
      else { v = Math.sin(wx * 0.075 + 1.6 * Math.sin(wy * 0.04)); k = smooth(0.35, 0.8, v) * smooth(0.95, 0.55, ny); }
      return [1, k * 0.9];
    }
    if (coat.patches) {
      var cp = CALICO[(coat.seed || 0) % CALICO.length];
      if (leg) return [0, 0];
      var fn = head ? cp.head : n === 'tail' ? cp.tail : n === 'torso' ? cp.torso : null;
      return fn ? fn(nx, ny, P) : [0, 0];
    }
    if (coat.tux) {
      var w = 0;
      if (n === 'torso') w = Math.max(smooth(1, 0.7, Math.hypot((nx - 0.86) / 0.24, (ny - 0.62) / 0.42)), smooth(0.8, 0.95, ny) * smooth(0.45, 0.7, nx));
      else if (leg) w = smooth(0.74, 0.82, ny);
      else if (n === 'headA' || n === 'headB') w = smooth(1, 0.75, Math.hypot((nx - 0.84) / 0.3, (ny - 0.78) / 0.26));
      else if (head) w = smooth(1, 0.75, Math.hypot((nx - 0.66) / 0.3, (ny - 0.8) / 0.22));
      return [1, w];
    }
    return [0, 0];
  }
  function tinted(coat) {
    if (tintCache[coat.key]) return tintCache[coat.key];
    var cols = [coat.base].concat(coat.stripe ? [coat.stripe] : coat.patches ? coat.patches : coat.tux ? [coat.tux] : []);
    var hsl = cols.map(function (c) { var q = rgbHsl(hexRgb(c)); q[2] = clamp(q[2], 0.06, 0.95); return q; });
    var blobs = null, out = {};
    PART_NAMES.forEach(function (n) {
      var b = base[n]; if (!b) return;
      if (!b.px || !b.m) { out[n] = b.img; return; }
      var P = C.parts[n] || {}, L0 = P.furL || GEO.cat.furL || 0.8, ox = P.x || 0, oy = P.y || 0, pat = cols.length > 1;
      var cv = document.createElement('canvas'); cv.width = b.w; cv.height = b.h;
      var c = cv.getContext('2d'), id = c.createImageData(b.w, b.h), s = b.px.data, d = id.data, m = b.m.data;
      for (var i = 0, p = 0; i < s.length; i += 4, p++) {
        d[i] = s[i]; d[i + 1] = s[i + 1]; d[i + 2] = s[i + 2]; d[i + 3] = s[i + 3];
        var k = m[i + 3] / 255;
        if (k <= 0.01) continue;
        var mx = Math.max(s[i], s[i + 1], s[i + 2]), mn = Math.min(s[i], s[i + 1], s[i + 2]), l = (mx + mn) / 510;
        var q = hsl[0], mix = 0, q2 = null;
        if (pat) {
          var x = p % b.w, y = (p / b.w) | 0, pa = patternAt(coat, n, x, y, ox + x, oy + y, blobs);
          if (pa[0] && pa[1] > 0.01) { q2 = hsl[pa[0]]; mix = pa[1]; }
        }
        var rgb = shadeFur(q, l, L0);
        if (q2) { var r2 = shadeFur(q2, l, L0); rgb = [lerp(rgb[0], r2[0], mix), lerp(rgb[1], r2[1], mix), lerp(rgb[2], r2[2], mix)]; }
        d[i] = s[i] + (rgb[0] - s[i]) * k; d[i + 1] = s[i + 1] + (rgb[1] - s[i + 1]) * k; d[i + 2] = s[i + 2] + (rgb[2] - s[i + 2]) * k;
      }
      c.putImageData(id, 0, 0);
      out[n] = cv;
    });
    tintCache[coat.key] = out; tintKeys.push(coat.key);
    if (tintKeys.length > 40) delete tintCache[tintKeys.shift()];
    return out;
  }
  function shadeFur(q, l, L0) {
    var Lt = q[2], nl = l <= L0 ? Lt * (l / L0) : Lt + (1 - Lt) * ((l - L0) / (1 - L0)) * (Lt > 0.85 ? 1 : 0.9);
    return hslRgb(q[0], q[1], clamp(nl, 0.02, 0.985));
  }

    return L > 0.68 ? '#5a3424' : '#ffffff';
  }

  function fit() {
    if (!built) return;
    var W = window.innerWidth || STAGE_W, H = window.innerHeight || STAGE_H;
    var s = Math.min(W / STAGE_W, H / STAGE_H) * num(F.globalScale, 100) / 100;

    var x = ax === 'left' ? 0 : ax === 'right' ? W - STAGE_W * s : (W - STAGE_W * s) / 2;

    el.stage.style.transform = 'translate(' + x + 'px,' + y + 'px) scale(' + s + ')';
    // viewport edges -> stage px -> machine px (the inverse of the stage + wrap transforms)
    var v = lay.view = { sL: -x / s, sR: (W - x) / s, sT: -y / s, sB: (H - y) / s };
    lay.vL = (v.sL - lay.x) / lay.k; lay.vR = (v.sR - lay.x) / lay.k; lay.vT = (v.sT - lay.y) / lay.k; lay.vB = (v.sB - lay.y) / lay.k;

  // where the cats come from: 'bottom' (leap up from under the screen edge), 'left' / 'right' (walk in along the
  // floor), 'auto' = the side with the most room between the box and the screen edge (bottom if neither has room)
  function route() {
    var mode = str(F.catEntrance, 'bottom'), cw = catW();
    var spL = B.x - lay.vL, spR = lay.vR - (B.x + B.w);
    if (mode === 'auto') mode = Math.max(spL, spR) < cw * 0.95 ? 'bottom' : spL >= spR ? 'left' : 'right';
    else if (mode === 'left' && spL < cw * 0.95) mode = spR >= cw * 0.95 ? 'right' : 'bottom';     // no room on that side
    else if (mode === 'right' && spR < cw * 0.95) mode = spL >= cw * 0.95 ? 'left' : 'bottom';
    var r = { mode: mode };
    if (mode === 'left' || mode === 'right') {
      var d = mode === 'left' ? -1 : 1;
      r.side = d;
      r.jump = mode === 'left' ? B.x - cw * 0.32 : B.x + B.w + cw * 0.32;
      r.off = mode === 'left' ? lay.vL - cw * 1.0 : lay.vR + cw * 1.0;
      r.slots = 0;
      for (var i = 0; i < 4; i++) {                   // waiting spots that are (mostly) on screen
        var wx = r.jump + d * cw * (0.62 + i * 0.56);
        if (d < 0 ? wx - cw * 0.3 < lay.vL : wx + cw * 0.3 > lay.vR) break;
        r.slots++;
      }
      if (!on(F.showLine, true)) r.slots = 0;
    } else {
      r.side = 0; r.slots = 0;
      r.under = Math.max(lay.vB, FLOOR) + catHt() * 1.25 + 170;  // fully below the screen edge, name tag + lifted tag included
    }
    return r;
  }

  /* ------------------------------------------------------------------ saved data (se_core BLOCK A, store v4)
     Pattern taken from working widgets: StreamElements/widgets (CommandCounters, StreamMarathon) + c4ldas
     top-monthly-subgifters. get once -> nothing saved yet (null) -> write the defaults right away -> every change
     writes the whole state as a plain JSON object (1 s throttle). Two safety belts: the widget never waits more than
     6 s for the store (it starts empty and asks once more after 10 s), and nothing is written to the store before
     its first answer, so a slow answer can't be overwritten. StreamElements overlays block localStorage: it is only
     used by the Live Preview (no SE_API there). STORE_KEY: letters/digits only. */
  var OLD_KEY = null;                                   // set to the previous letters/digits key if STORE_KEY ever changes
  var svReady = false, svStarted = false, svPending = null, svTimer = 0, svLast = null;
  var svInfo = { asked: 0, answered: false, how: '', sets: 0, err: '' };

  function svHasSE() { return typeof SE_API !== 'undefined' && SE_API && SE_API.store && typeof SE_API.store.get === 'function'; }
  function svObj(o) {                                   // also opens older { format: 2, data: '<json>' } saves
    if (o && typeof o === 'object' && o.format === 2 && typeof o.data === 'string') { try { o = JSON.parse(o.data); } catch (e) { o = null; } }
    return o && typeof o === 'object' && !Array.isArray(o) ? o : null;
  }
  function svGet(k) { try { return Promise.resolve(SE_API.store.get(k)).then(svObj, function () { return null; }); } catch (e) { return Promise.resolve(null); } }

  // cb(saved | null) is called exactly once per load. late(saved) only runs if the store answers after the 6 s wait
  // with real data: adopt it, redraw and call saveStore() so it (not the empty start) is what gets written.
  function loadStore(cb, late) {
    if (svStarted) return cb(svLast);                   // onWidgetLoad again (editor): keep what we already hold
    svStarted = true;
    if (!svHasSE()) {                                   // Live Preview / local test
      var loc = null; try { loc = svObj(JSON.parse(localStorage.getItem(STORE_KEY))); } catch (e) {}
      svReady = true; svInfo.how = 'preview'; svLast = loc; return cb(loc);
    }
    var done = false;
    var t = setTimeout(function () {
      if (done) return;
      done = true; svInfo.how = 'slow, started empty'; console.warn('[widget] StreamElements store is slow, starting empty');
      cb(null);
    }, 6000);
    function answer(d, how) {
      if (svInfo.answered) return;
      svInfo.answered = true; svInfo.how = how; svReady = true;
      if (d) svLast = d;
      if (!done) { done = true; clearTimeout(t); cb(d); }
      else if (d && late) late(d);                       // a real save arrived late: adopt it
      if (svPending) svFlush();                          // a write that waited for the answer goes out now
    }
    function ask(retry) {
      svInfo.asked++;
      svGet(STORE_KEY).then(function (d) {
        if (d || !OLD_KEY) return answer(d, d ? 'loaded' : 'new key, defaults saved');
        return svGet(OLD_KEY).then(function (o) { answer(o, o ? 'moved from ' + OLD_KEY : 'new key, defaults saved'); });
      });
      if (!retry) setTimeout(function () { if (!svInfo.answered) ask(true); }, 10000);   // one retry
    }
    ask(false);
  }

  function svFlush() {
    clearTimeout(svTimer); svTimer = 0;
    var obj = svPending; if (!obj) return;
    try { localStorage.setItem(STORE_KEY, JSON.stringify(obj)); } catch (e) {}           // preview only; blocked in the overlay
    if (!svReady || !svHasSE()) return;                                                  // never write before the store answered
    svPending = null; svWroteAt = Date.now();
    try {
      Promise.resolve(SE_API.store.set(STORE_KEY, JSON.parse(JSON.stringify(obj))))      // plain JSON copy
        .then(function () { svInfo.sets++; svInfo.err = ''; }, function (e) { svInfo.err = String((e && e.message) || e); });
    } catch (e) { svInfo.err = String((e && e.message) || e); }
  }
  // A lone change is written at once; a burst becomes one write per second; whatever is still waiting is sent the
  // moment the page goes away. (A 1 s delay on every write lost the last change when OBS closed or the editor was
  // refreshed right after an event - found live 2026-10-09.)
  var svWroteAt = 0;
  function saveStore(obj, now) {                        // give it the WHOLE state on every change
    svPending = obj; svLast = obj;
    var wait = 1000 - (Date.now() - svWroteAt);
    if (now || (!svTimer && wait <= 0)) return svFlush();
    if (!svTimer) svTimer = setTimeout(svFlush, Math.max(0, wait));
  }
  ['pagehide', 'beforeunload'].forEach(function (ev) { window.addEventListener(ev, function () { if (svPending) svFlush(); }); });
  document.addEventListener('visibilitychange', function () { if (document.hidden && svPending) svFlush(); });

  Cat.prototype.hop = async function (x, y, top, layer, back) {

      var up = k < ta ? 1 - k / ta : 0;                      // stretch while rising
      pose.sx = 1 - 0.07 * up; pose.sy = 1 + 0.1 * up;
      if (!switched && k >= ta) { switched = true; self.setLayer(layer, back); }
    });
    pose.air = 0; pose.lean = 0; pose.crouch = 0; this.setLayer(layer, back);
    if (this.y < lay.vB) { play('land'); dust(this.x, this.y, layer === el.inside); }
    await tween(200, function (k) { var e = Math.sin(k * Math.PI); pose.sx = 1 + 0.12 * e; pose.sy = 1 - 0.13 * e; });
    pose.sx = 1; pose.sy = 1;
  };
  Cat.prototype.sniff = async function (ms) {
    var self = this;
    await tween(180, function (k) { self.pose.sniff = k; });
    var p = this.headTop(); bubble('?', p[0] + this.dir * 70, p[1] - 30);
    await sleep(ms || 650);
    await tween(160, function (k) { self.pose.sniff = 1 - k; });
  };

  /* ------------------------------------------------------------------ orders: several cats at once, each in its own lane
     Up to "Cats in the box at once" (default 3) visits run side by side, each in its own lane with its own timing.
     Lanes alternate back / front (back = a little higher and drawn behind), so cats never stand in each other.
     Extra events wait in line next to the box (side entrances) or as a "+N" bubble. The scoop waits until every
     active visit is done, and no new cat starts while it is pending. */
  var demoEv = false, active = [], lanes = [], cleaning = false, nextStart = 0, pumpTimer = 0;
  function maxIn() { return clamp(Math.round(num(F.maxInBox, 3)), 1, 4); }
  function isBusy() { return active.length > 0 || cleaning; }
  function laneSpots(n) {
    var a = lay.inMin, b = lay.inMax, out = [];
    for (var i = 0; i < n; i++) {
      var x = n === 1 ? (a + b) / 2 : lerp(a - catW() * 0.12, b + catW() * 0.12, i / (n - 1));
      var back = n > 1 && i % 2 === 1;
      out.push({ i: i, x: clamp(x, B.x + B.innerX[0] + catW() * 0.3, B.x + B.innerX[1] - catW() * 0.3), back: back,
                 feet: INSIDE_FEET + (back ? -26 : 4), z: back ? 1 : 3, scale: n >= 4 ? 0.72 : n === 3 ? 0.82 : n === 2 ? 0.92 : 1 });
    }
    return out;
  }
  function takeLane(prefer) {
    var n = maxIn(), spots = laneSpots(n), free = spots.filter(function (s) { return !lanes[s.i]; });
    if (!free.length) return null;
    free.sort(function (p, q) { return Math.abs(p.x - prefer) - Math.abs(q.x - prefer) || (p.back - q.back); });
    lanes[free[0].i] = true;
    return free[0];
  }
  function freeLane(l) { if (l) lanes[l.i] = false; }
  function kindFor(o) {
    if (!on(F.catCharacters, true)) return '';
    if (o.source === 'follow') return 'kitten';
    if (o.source === 'sub') return 'crown';
    if ((o.source === 'tip' && o.amount >= num(F.goldenTip, 10)) || (o.source === 'cheer' && o.amount >= num(F.goldenTip, 10) * 100)) return 'swag';
    return '';
  }
  function newCat(o) {
    var kind = kindFor(o), c = new Cat(o.coat, kind === 'swag' ? '' : kind, o.tagColor);
    if (kind === 'swag') { c.pose.swag = 1; c.pose.tailUp = 0.6; c.pose.headTilt = -5; }
    c.name = o.name; c.tagSync();
    return c;
  }
  function order(o) {
    if (demoEv) o.demo = true;
    o.name = str(o.name, '').trim() || 'Someone';
    o.tagColor = eventHex(o.source);
    o.coat = coatFor(o.login || o.name, o.source);
    if (queue.length >= clamp(num(F.maxQueue, 20), 1, 99)) { if (o.goal) addGoal(o.goal); return; }
    if (o.goal) o.held = addGoal(o.goal, true);
    stopPeek();
    queue.push(o);
    wake();
    layoutQueue();
    pump();
  }
  // side entrances: cats that arrive close together start spaced out along the walk, each with its own tag height,
  // so names never stack from the very first step (QA tur 3)
  var spawnAt = 0, spawnK = 0;
  function spawnSide(cat, r) {
    var now = Date.now();
    spawnK = now - spawnAt < 2600 ? (spawnK + 1) % 3 : 0; spawnAt = now;
    cat.x = r.off + r.side * spawnK * catW() * 0.85; cat.y = FLOOR; cat.placed = true; cat.tagLift = spawnK * 48;
    cat.tick(performance.now() / 1000, 0);
  }
  function layoutQueue() {
    if (!built || !base.torso || !lay.view) return;
    var r = route(), shown = 0;
    for (var i = 0; i < queue.length; i++) {
      var o = queue[i];
      if (i < r.slots) {
        if (!o.cat) { o.cat = newCat(o); spawnSide(o.cat, r); }
        o.cat.scale = 0.86;
        o.cat.walkTo(waitX(r, i), FLOOR).then(function (c) { return function () { if (!c.dead && !c.serving) c.dir = -r.side; }; }(o.cat));
        shown++;
      }
    }
    var more = queue.length - shown;
    el.more.hidden = !(more > 0);
    el.more.textContent = '+' + more;
    var edge = r.side ? r.side : (lay.vR - (B.x + B.w) > B.x - lay.vL ? 1 : -1);
    px(el.more, edge < 0 ? B.x + 70 : B.x + B.w - 70, B.y + 20);
  }
  function pump() {
    if (!ready || !built) return;
    if (!base.torso) { catReady.then(pump); return; }
    clearTimeout(pumpTimer);
    if (cleaning) return;
    if (pendingClean) {
      if (active.length) return;                                  // let every cat in the box finish first
      var why = pendingClean; pendingClean = false; cleaning = true;
      wake();                                                     // the scoop / goal celebration is visible
      stopPeek().then(function () { return clean(why); }).catch(function (e) { console.error('[litter buddy]', e); })
        .then(function () { cleaning = false; lastBusy = Date.now(); pump(); if (!lbBusy()) armHide(); });
      return;
    }
    while (queue.length && active.length < maxIn()) {
      var now = Date.now();
      if (now < nextStart) { pumpTimer = setTimeout(pump, nextStart - now + 10); return; }
      var o = queue[0], lane = takeLane((lay.inMin + lay.inMax) / 2);
      if (!lane) return;
      queue.shift();
      nextStart = now + (route().side ? 1500 : 650) / speed();   // side entrances share one doorway
      startVisit(o, lane);
    }
  }
  function startVisit(o, lane) {
    var v = { o: o, lane: lane, extra: [] };
    active.push(v);
    fastMode = active.some(function (q) { return q.o.fast; });
    if (peeker) stopPeek();
    serve(o, v).catch(function (e) { console.error('[litter buddy]', e); if (o.held) { showGoal(o.held); o.held = 0; } if (o.cat) o.cat.remove(); })
      .then(function () {
        active.splice(active.indexOf(v), 1);
        freeLane(lane); v.extra.forEach(freeLane);
        fastMode = active.some(function (q) { return q.o.fast; });
        lastBusy = Date.now();
        pump();
        if (!lbBusy()) armHide();
      });
  }

  // ------------------------------------------------------------------ one visit
  async function serve(o, v) {
    layout(); fit();
    var r = route(), lane = v.lane;
    var cat = o.cat || newCat(o);
    o.cat = cat; cat.serving = true;
    layoutQueue();
    var s0 = cat.scale; tween(400, function (k) { cat.scale = lerp(s0, lane.scale, k); });
    state.orders++; save();
    o.no = state.orders;
    var type = poopType(o), inX = lane.x, feet = lane.feet;
    // raid party: friends take the free lanes, hop in, watch the leader and cheer
    var party = [];
    if (o.source === 'raid' && on(F.raidParty, true)) {
      var want = o.amount >= 50 ? 4 : o.amount >= 10 ? 3 : 2;
      for (var i = 0; i < want; i++) {
        var fl = takeLane(inX + (i % 2 ? 1 : -1) * catW());
        if (!fl) break;
        v.extra.push(fl);
        var f = new Cat(coatFor(o.name + '#' + (i + 1), 'raid'), '', o.tagColor); f.scale = fl.scale * 0.9; f.lane = fl;
        party.push(f);
      }
    }
    // 1) come to the box and hop in
    var apex = RIM - catHt() * 0.55;
    cat.el.style.zIndex = lane.z;
    if (r.mode === 'bottom') {
      var from = inX - catW() * 0.55;
      cat.x = from; cat.y = r.under; cat.setLayer(el.floor); cat.tick(performance.now() / 1000, 0);
      party.forEach(function (f, i) { f.x = f.lane.x - catW() * 0.5; f.y = r.under; f.tick(performance.now() / 1000, 0); });
      play('meow');
      await cat.hop(inX, feet, apex, el.inside);
      partyIn(party, true);
      await cat.sniff(500);
    } else {
      if (!cat.placed) spawnSide(cat, r);
      party.forEach(function (f, i) { f.x = r.off + r.side * (i + 1) * catW() * 0.45; f.y = FLOOR; f.tick(performance.now() / 1000, 0); f.walkTo(r.jump + r.side * (i + 1) * catW() * 0.42, FLOOR); });
      await cat.walkTo(r.jump + r.side * (lane.i % 3) * catW() * 0.42, FLOOR);
      cat.dir = -r.side;
      await cat.sniff(650);
      play('meow');
      await cat.hop(inX, feet, apex, el.inside);
      cat.tagLift = 0;
      partyIn(party, false);
    }
    cat.el.style.zIndex = lane.z;
    // 2) dig a little hole
    cat.pose.dig = 1; play('dig', 1300);
    var dg = setInterval(function () { var p = cat.at(C.parts.legB.x + C.parts.legB.w * 0.6, C.feet - 30); spray(p[0], Math.min(p[1], RIM - 8), 3, -cat.dir, 1); }, 90);
    await sleep(1300);
    clearInterval(dg); cat.pose.dig = 0;
    // 3) squat (straining face) ... plop
    party.forEach(function (f) { f.dir = f.x < cat.x ? 1 : -1; f.pose.face = 'surprised'; });
    await tween(260, function (k) { cat.pose.squat = k; cat.pose.tailUp = Math.max(cat.pose.tailUp, k); });
    await sleep(700);
    var rp = cat.at(C.parts.torso.x + C.parts.torso.w * 0.08, C.parts.torso.y + C.parts.torso.h * 0.72);
    dropPoop(type, rp, o);
    if (o.held) { var h = o.held; o.held = 0; showGoal(h); }
    showReceipt(o, type);
    party.forEach(function (f, i) { f.pose.face = ''; f.pose.happy = 1; setTimeout(function () { hearts(f); }, i * 150); });
    await sleep(500);
    await tween(240, function (k) { cat.pose.squat = 1 - k; cat.pose.tailUp = (cat.pose.swag ? 0.6 : 0) + (1 - k) * (cat.pose.swag ? 0.4 : 1); });
    // 4) kick sand over it
    cat.pose.kick = 1; play('dig', 900);
    var kk = setInterval(function () { var p = cat.at(C.parts.legH.x + C.parts.legH.w * 0.3, C.feet - 40); spray(p[0], Math.min(p[1], RIM - 8), 3, -cat.dir, 0.8); }, 80);
    await sleep(900);
    clearInterval(kk); cat.pose.kick = 0;
    // 5) proud moment, then out: shake a paw, tail up, off it goes
    cat.pose.happy = 1; play('meow'); hearts(cat);
    await sleep(520);
    cat.pose.happy = 0;
    var outs = [];
    if (r.mode === 'bottom') {
      await shakePaw(cat);
      proud(cat);
      party.forEach(function (f, i) {
        outs.push(sleep(150 + i * 160).then(function () { f.pose.happy = 0; return f.hop(f.x + catW() * 0.5, r.under, apex, el.floor); }).then(function () { f.remove(); }));
      });
      await cat.hop(inX + catW() * 0.6, r.under, apex, el.floor);
      cat.remove();
    } else {
      var ex = r.jump + r.side * catW() * 0.2;
      party.forEach(function (f, i) {
        outs.push(sleep(120 + i * 180).then(function () { f.pose.happy = 0; return f.hop(ex + r.side * (i + 1) * catW() * 0.35, FLOOR, apex, el.floor); })
          .then(function () { proud(f); return f.walkTo(r.off + r.side * catW() * (i + 1) * 0.4, FLOOR, 1.25); }).then(function () { f.remove(); }));
      });
      await cat.hop(ex, FLOOR, apex, el.floor);
      cat.el.style.zIndex = '';
      await shakePaw(cat);
      proud(cat);
      cat.walkTo(r.off, FLOOR, 1.3).then(function () { cat.remove(); });
      await sleep(350);
    }
    await Promise.all(outs);
  }
  function partyIn(party, fromBelow) {
    party.forEach(function (f, i) {
      setTimeout(function () {
        var go = fromBelow ? Promise.resolve() : f.walkTo(f.x, FLOOR);
        go.then(function () { f.el.style.zIndex = f.lane.z; return f.hop(f.lane.x, f.lane.feet, RIM - catHt() * 0.5, el.inside); })
          .then(function () { f.el.style.zIndex = f.lane.z; f.dir = -f.dir; });
      }, 260 + i * 220);
    });
  }
  async function shakePaw(cat) {

  function proud(cat) { cat.pose.tailUp = 1; cat.pose.headTilt = -7; cat.pose.face = 'smug'; }
  function poopType(o) {
    var f = Math.round(num(F.forcePoop, 0));
    if (f >= 1 && f <= 4) return f - 1;
    var big = num(F.goldenTip, 10);
    if (o.source === 'tip' && o.amount >= big) return 2;
    if (o.source === 'cheer' && o.amount >= big * 100) return 2;
    if (o.source === 'sub' || o.source === 'raid') return 3;
    return Math.random() < 0.5 ? 0 : 1;
  }

    play('plop');

    var sl = slots(), idx = full ? cap - 1 - Math.floor(Math.random() * Math.min(cap, 5)) : list.length - 1;

    d.animate([

      { transform: 'translate(0,0) scale(1.18,.78)', offset: 0.8 },

    setTimeout(function () {
      puffs(to.x, to.y - h * 0.9, type === 2 ? 0 : 3);

      if (on(F.autoScoop, true)) { pendingClean = pendingClean || 'full'; }
    } else {
      setTimeout(function () { renderPile(); }, 600 / speed());
    }
    return to;
  }

  /* ------------------------------------------------------------------ the idle peeker: a cat peeks over the rim now and then */
  var peeker = null, nextPeek = 0, lastBusy = 0;
  function maybePeek(now) {
    if (!on(F.idlePeek, true) || isBusy() || queue.length || peeker || !ready || !base.torso || cats.length) return;
    if (!nextPeek) { nextPeek = now + 9000 + Math.random() * 8000; return; }
    if (now < nextPeek || now - lastBusy < 6000) return;
    nextPeek = now + clamp(num(F.peekEvery, 40), 10, 600) * 1000 * (0.7 + Math.random() * 0.6);
    peek();
  }
  async function peek() {
    var c = new Cat(coatFor('peek' + Math.floor(Math.random() * 1e6), 'chat'), '');
    peeker = c; c.scale = 0.9; c.dir = Math.random() < 0.5 ? 1 : -1;
    c.x = lerp(lay.inMin, lay.inMax, 0.2 + Math.random() * 0.6);
    var low = RIM + catHt() * 1.02, up = RIM + catHt() * 0.5;
    c.y = low; c.setLayer(el.inside); c.pose.face = 'sleepy'; c.tick(performance.now() / 1000, 0);
    await tween(700, function (k) { if (peeker === c) c.y = lerp(low, up, easeOut(k)); });
    if (peeker !== c) return;
    c.pose.face = 'A';
    await sleep(1200);
    if (peeker === c) { c.dir = -c.dir; await sleep(1100); }
    if (peeker === c) { c.pose.face = 'sleepy'; await sleep(700); }
    if (peeker === c) await stopPeek();
  }
  function stopPeek() {
    var c = peeker; if (!c) return Promise.resolve();
    peeker = null;
    var y0 = c.y, low = RIM + catHt() * 1.05;
    return tween(300, function (k) { c.y = lerp(y0, low, k); }).then(function () { c.remove(); });
  }

  /* ------------------------------------------------------------------ the big scoop (goal reached / box full) */
  async function clean(why) {
    var celebrateGoal = why === 'goal';
    el.more.style.opacity = '0';
    setTimeout(function () { el.more.style.opacity = ''; }, 4200 / speed());
    var watchers = cats.filter(function (c) { return !c.serving && !c.dead && c !== peeker && c.layer === el.floor; });

      tab.multi = true;
      tab.rows.forEach(function (r) { r.el.remove(); });
      tab.rows = tab.entries.slice(0, -1).filter(function (e) { return !e.gone; }).map(function (e) { var r = { e: e, el: rowFor(e, true) }; tab.lines.appendChild(r.el); return r; });
    }
    var row = { e: entry, el: rowFor(entry, multi) };

    switch (o.source) {

      case 'follow': return 'FOLLOW <3';

  function barcode(seed) {
    var h = 2166136261;
    for (var i = 0; i < seed.length; i++) h = Math.imul(h ^ seed.charCodeAt(i), 16777619);
    function rnd() { h ^= h << 13; h ^= h >>> 17; h ^= h << 5; return ((h >>> 0) % 1000) / 1000; }
    var x = 0, bar = true, rects = '', digits = '';
    while (x < 296) {
      var w = 1 + Math.floor(rnd() * 4);
      if (bar) rects += '<rect x="' + x + '" y="0" width="' + w + '" height="60"/>';
      x += w; bar = !bar;
    }
    for (i = 0; i < 12; i++) digits += Math.floor(rnd() * 10);
    return { svg: '<svg viewBox="0 0 300 60" preserveAspectRatio="none">' + rects + '</svg>', digits: digits };
  }

  var AC = null, noiseBuf = null;
  function volume() { return clamp(num(F.volume, 60), 0, 100) / 100; }
  function audio() {
    try {
      if (!AC) { var Ctor = window.AudioContext || window.webkitAudioContext; if (!Ctor) return null; AC = new Ctor(); }
      if (AC.state === 'suspended') AC.resume();
    } catch (e) { return null; }
    return AC;
  }
  function noise(c) {
    if (!noiseBuf) {
      noiseBuf = c.createBuffer(1, c.sampleRate * 2, c.sampleRate);
      var d = noiseBuf.getChannelData(0);
      for (var i = 0; i < d.length; i++) d[i] = Math.random() * 2 - 1;
    }
    var s = c.createBufferSource(); s.buffer = noiseBuf; s.loop = true; return s;
  }
  function play(name, ms) {
    if (!on(F.soundOn, true) || volume() <= 0) return;
    var field = { meow: 'soundMeow', dig: 'soundDig', plop: 'soundPlop', done: 'soundDone', receipt: 'soundReceipt' }[name];
    var url = field ? str(F[field], '').trim() : '';
    if (url) {
      try {
        var a = new Audio(url); a.volume = volume();
        var p = a.play(); if (p && p.catch) p.catch(function () {});
        if (ms) setTimeout(function () { fadeOut(a); }, ms);
      } catch (e) {}
      return;
    }
    var c = audio(); if (!c) return;
    try { SYNTH[name](c, c.currentTime + 0.02, (ms || 0) / 1000); } catch (e) {}
  }
  function fadeOut(a) {
    var v = a.volume, n = 0, iv = setInterval(function () {
      n++; a.volume = Math.max(0, v * (1 - n / 10));
      if (n >= 10) { clearInterval(iv); a.pause(); }
    }, 40);
  }
  function tone(c, dest, t, f, type, peak, len) {
    var o = c.createOscillator(), g = c.createGain();
    o.type = type; o.frequency.value = f;
    g.gain.setValueAtTime(0.0001, t); g.gain.exponentialRampToValueAtTime(peak, t + 0.01); g.gain.exponentialRampToValueAtTime(0.0001, t + len);
    o.connect(g); g.connect(dest); o.start(t); o.stop(t + len + 0.02);
  }
  var SYNTH = {
    meow: function (c, t) {                      // a short cartoon "mew": a gliding nasal tone through two formants
      var f0 = 520 + Math.random() * 160, len = 0.42, out = c.createGain();
      out.gain.setValueAtTime(0.0001, t); out.gain.exponentialRampToValueAtTime(volume() * 0.22, t + 0.05);
      out.gain.setValueAtTime(volume() * 0.22, t + len * 0.6); out.gain.exponentialRampToValueAtTime(0.0001, t + len);
      out.connect(c.destination);
      var o = c.createOscillator(); o.type = 'sawtooth';
      o.frequency.setValueAtTime(f0, t); o.frequency.linearRampToValueAtTime(f0 * 1.45, t + len * 0.35); o.frequency.linearRampToValueAtTime(f0 * 0.9, t + len);
      [[900, 4], [2300, 6]].forEach(function (fq) {
        var bp = c.createBiquadFilter(); bp.type = 'bandpass'; bp.Q.value = fq[1];
        bp.frequency.setValueAtTime(fq[0] * 0.7, t); bp.frequency.linearRampToValueAtTime(fq[0] * 1.2, t + len * 0.4); bp.frequency.linearRampToValueAtTime(fq[0] * 0.8, t + len);
        o.connect(bp); bp.connect(out);
      });
      o.start(t); o.stop(t + len + 0.05);
    },
    dig: function (c, t, dur) {                  // scratchy sand: short high noise bursts
      dur = Math.max(0.4, Math.min(dur || 1, 4));
      var out = c.createGain(); out.gain.value = volume() * 0.35; out.connect(c.destination);
      var n = noise(c), bp = c.createBiquadFilter(), g = c.createGain();
      bp.type = 'bandpass'; bp.frequency.value = 2600; bp.Q.value = 0.8; g.gain.value = 0;
      n.connect(bp); bp.connect(g); g.connect(out);
      for (var x = 0; x < dur; x += 0.09) { g.gain.setValueAtTime(0.0001, t + x); g.gain.linearRampToValueAtTime(0.6, t + x + 0.015); g.gain.linearRampToValueAtTime(0.0001, t + x + 0.07); }
      n.start(t); n.stop(t + dur + 0.05);
    },
    plop: function (c, t) {
      var out = c.createGain(); out.gain.value = volume() * 0.5; out.connect(c.destination);
      var o = c.createOscillator(), g = c.createGain();
      o.type = 'sine'; o.frequency.setValueAtTime(620, t); o.frequency.exponentialRampToValueAtTime(140, t + 0.14);
      g.gain.setValueAtTime(0.0001, t); g.gain.exponentialRampToValueAtTime(0.9, t + 0.01); g.gain.exponentialRampToValueAtTime(0.0001, t + 0.18);
      o.connect(g); g.connect(out); o.start(t); o.stop(t + 0.2);
      tone(c, out, t + 0.16, 1760, 'triangle', 0.18, 0.12);
    },
    hop: function (c, t) {
      var out = c.createGain(); out.gain.value = volume() * 0.2; out.connect(c.destination);
      var o = c.createOscillator(), g = c.createGain();
      o.type = 'triangle'; o.frequency.setValueAtTime(330, t); o.frequency.exponentialRampToValueAtTime(880, t + 0.16);
      g.gain.setValueAtTime(0.0001, t); g.gain.exponentialRampToValueAtTime(0.7, t + 0.02); g.gain.exponentialRampToValueAtTime(0.0001, t + 0.2);
      o.connect(g); g.connect(out); o.start(t); o.stop(t + 0.22);
    },
    land: function (c, t) {
      var out = c.createGain(); out.gain.value = volume() * 0.3; out.connect(c.destination);
      var n = noise(c), lp = c.createBiquadFilter(), g = c.createGain();
      lp.type = 'lowpass'; lp.frequency.value = 900;
      g.gain.setValueAtTime(0.0001, t); g.gain.exponentialRampToValueAtTime(0.8, t + 0.01); g.gain.exponentialRampToValueAtTime(0.0001, t + 0.16);
      n.connect(lp); lp.connect(g); g.connect(out); n.start(t); n.stop(t + 0.2);
    },
    swoosh: function (c, t) {
      var out = c.createGain(); out.gain.value = volume() * 0.35; out.connect(c.destination);
      var n = noise(c), bp = c.createBiquadFilter(), g = c.createGain();
      bp.type = 'bandpass'; bp.Q.value = 1.2; bp.frequency.setValueAtTime(400, t); bp.frequency.exponentialRampToValueAtTime(2400, t + 0.5);
      g.gain.setValueAtTime(0.0001, t); g.gain.exponentialRampToValueAtTime(0.7, t + 0.2); g.gain.exponentialRampToValueAtTime(0.0001, t + 0.6);
      n.connect(bp); bp.connect(g); g.connect(out); n.start(t); n.stop(t + 0.65);
    },
    chime: function (c, t) {
      var out = c.createGain(); out.gain.value = volume() * 0.28; out.connect(c.destination);
      [[880, 0], [1318, 0.09]].forEach(function (p) { tone(c, out, t + p[1], p[0], 'triangle', 0.6, 0.35); });
    },
    done: function (c, t) {
      var out = c.createGain(); out.gain.value = volume() * 0.32; out.connect(c.destination);
      [[523, 0], [659, 0.1], [784, 0.2], [1046, 0.32]].forEach(function (p) { tone(c, out, t + p[1], p[0], 'triangle', 0.7, 0.35); });
    },
    receipt: function (c, t) {
      var dur = 1.3, out = c.createGain(); out.gain.value = volume() * 0.4; out.connect(c.destination);
      var n = noise(c), hp = c.createBiquadFilter(), g = c.createGain();
      hp.type = 'highpass'; hp.frequency.value = 2200; g.gain.value = 0;
      n.connect(hp); hp.connect(g); g.connect(out);
      for (var x = 0; x < dur; x += 0.032) { g.gain.setValueAtTime(0.35 + Math.random() * 0.5, t + x); g.gain.setValueAtTime(0.02, t + x + 0.012); }
      g.gain.setValueAtTime(0, t + dur);
      n.start(t); n.stop(t + dur + 0.05);
    }
  };

  /* Twitch channel points. Current StreamElements sends a redemption on the generic 'event' listener as
     { type: 'channelPointsRedemption', data: { username, displayName, redemption, message, amount } }
     (older shapes: listener 'channel-points-latest', or type 'channelPoints' with data.title). cpNormalize reshapes
     any of them into { type: 'channelPoints', data } for the handler below, drops the same redemption arriving on a
     second listener, and matches the reward name loosely ("Brew a Potion 🧪" == "brew a potion").
     A plain type 'redemption' is the StreamElements loyalty store, not channel points, and is left alone. */
  function rewardKey(v) {
    return str(v, '').toLowerCase().replace(/[^0-9a-zÀ-ɏЀ-ӿ぀-ヿ一-鿿가-힯]+/g, '');
  }
  // One redemption can reach the widget up to 3 ways (event / channel-points-latest / session data). A copy is the
  // same viewer + text arriving on a path that hasn't delivered it yet (within 10 s). The same path delivering it
  // again means the viewer really redeemed twice, and both count.
  var redeems = [];
  function isMirror(user, text, src) {
    var now = Date.now(), key = (str(user, '') + '\u0001' + str(text, '')).toLowerCase();
    redeems = redeems.filter(function (r) { return now - r.at < 10000; });
    for (var i = 0; i < redeems.length; i++) {
      var r = redeems[i];
      if (r.key === key && !r.src[src]) { r.src[src] = 1; return true; }
    }
    var o = { key: key, at: now, src: {} }; o.src[src] = 1;
    redeems.push(o);
    return false;
  }
  function cpNormalize(L, e) {
    var t = String((e && e.type) || '');
    if (t !== 'channelPointsRedemption' && t !== 'channelPoints' && t !== 'channel-points' && L !== 'channel-points-latest') return null;
    var d = e.data || {};
    var title = str(d.redemption || d.title || d.rewardTitle || (d.reward && d.reward.title) || e.redemption || '', '');
    var user = str(d.displayName || d.username || d.userName || d.user_name || d.name || e.name || '', '');
    var text = str(d.message || d.userInput || d.user_input || d.text || e.message || '', '');
    if (isMirror(user, text, e._session ? 'session' : L === 'channel-points-latest' ? 'latest' : 'event')) return false;
    [F.rewardName, F.rewardStart, F.rewardBoost].forEach(function (w) {
      w = str(w, '').trim();
      if (w && rewardKey(w) === rewardKey(title)) title = w;       // the handler compares against the field text
    });
    return { type: 'channelPoints', data: { title: title, redemption: title, displayName: user || 'Someone', username: user || 'Someone',
      userInput: text, message: text, amount: num(d.amount || d.cost || e.amount, 0) } };
  }
  // Fallback: a redemption that only shows up in session data is replayed through onEventReceived once.
  var cpLast = '';
  function cpSig(cp) { return [cp.name, cp.redemption, cp.message, cp.amount].join('\u0001').toLowerCase(); }
  window.addEventListener('onWidgetLoad', function (obj) {
    var s = obj && obj.detail && obj.detail.session && obj.detail.session.data;
    var cp = s && s['channel-points-latest'];
    if (cp && typeof cp === 'object') cpLast = cpSig(cp);          // don't replay the last redemption on load
  });
  window.addEventListener('onSessionUpdate', function (obj) {
    var cp = obj && obj.detail && obj.detail.session && obj.detail.session['channel-points-latest'];
    if (!cp || typeof cp !== 'object' || !cp.redemption) return;
    var sig = cpSig(cp);
    if (sig === cpLast) return;
    cpLast = sig;
    window.dispatchEvent(new CustomEvent('onEventReceived', { detail: { listener: 'channel-points-latest',
      event: { name: cp.name, amount: cp.amount, message: cp.message, redemption: cp.redemption, _session: true } } }));
  });

  // Live StreamElements wraps some events. A button click in the widget settings arrives as listener 'event:test'
  // with { listener: 'widget-button', field, value } inside, and the overlay editor's Emulate panel sends 'event:test'
  // (or 'event' with isMock) carrying { listener, event } or a bare { type: 'tip', data }. Unwrap them to the normal shape.
  // A live event mirrored on plain 'event' is left alone: its own *-latest listener already handles it.
  var TYPE_MAP = { follower: 'follower-latest', follow: 'follower-latest', subscriber: 'subscriber-latest', sub: 'subscriber-latest',
    sponsor: 'sponsor-latest', cheer: 'cheer-latest', tip: 'tip-latest', donation: 'tip-latest', superchat: 'superchat-latest',
    raid: 'raid-latest', kicks: 'kicks-latest', kick: 'kicks-latest' };
  function unwrapTest(L, e) {
    e = e || {};
    if (L !== 'event' && L !== 'event:test') return [L, e];
    if (e.listener === 'widget-button') return ['widget-button', (e.event && e.event.field) ? e.event : e];
    if (L === 'event' && !e.isMock) return [L, e];
    if (e.listener && e.listener !== L) return [e.listener, e.event || e];
    var t = TYPE_MAP[str(e.type, '').toLowerCase()];
    if (!t) return [L, e];
    var dd = (e.data && typeof e.data === 'object') ? e.data : {}, o = {}, k;
    for (k in dd) o[k] = dd[k];
    for (k in e) if (k !== 'data' && e[k] !== undefined) o[k] = e[k];
    o.name = str(e.name || dd.displayName || dd.username || dd.name, 'Someone');
    o.amount = e.amount != null ? e.amount : dd.amount;
    o.message = str(e.message || dd.message, '');
    return [t, o];
  }

  // One alert can reach the widget twice (Emulate sends 'event:test' and the plain '*-latest' / isMock 'event' copy,
  // buyer 2026-10-25: two cats + two poops per test). A copy = same type + name + amount from a listener that hasn't
  // delivered it yet (3 s). The same live listener again is a real repeat and counts; Emulate ('event…' sources) can
  // send one alert twice on 'event:test' itself (wrapped + bare, "katine"/"Katine"), so a test repeat is always a copy.
  var alerts = [];
  function isAlertCopy(L, e, src) {
    var now = Date.now(), key = [L, str(e.name, ''), num(e.amount, 0)].join('\u0001').toLowerCase();
    alerts = alerts.filter(function (a) { return now - a.at < 3000; });
    for (var i = 0; i < alerts.length; i++) {
      var a = alerts[i];
      if (a.key === key && (!a.src[src] || /^event/.test(src))) { a.src[src] = 1; return true; }
    }
    var o = { key: key, at: now, src: {} }; o.src[src] = 1;
    alerts.push(o);
    return false;
  }

  }
